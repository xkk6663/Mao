$ErrorActionPreference = 'Stop'
$pwshExe = (Get-Command pwsh).Source
$startScript = 'C:\Users\xiao1\Mio\start_mio.ps1'
$desktop = [Environment]::GetFolderPath('Desktop')
$startup = [Environment]::GetFolderPath('Startup')
$icon = 'C:\Users\xiao1\Mio\澪Agent应用\desktop\mio.ico'

# ---------- 1) 桌面 一键启动桌宠.cmd (UTF-8 with BOM + chcp 65001) ----------
$cmdContent = @'
@echo off
chcp 65001 >nul
title Mio 桌宠一键启动
echo.
echo   ==========================================
echo     Mio 桌宠 一键启动（后端 + Live2D 桌宠）
echo   ==========================================
echo.
'@
$cmdContent += "`"$pwshExe`" -NoProfile -ExecutionPolicy Bypass -File `"$startScript`""
$cmdContent += @'

echo.
echo   启动完成！桌宠应已出现在屏幕上，本窗口可关闭。
echo.
pause
'@
$cmdPath = Join-Path $desktop '一键启动桌宠.cmd'
[IO.File]::WriteAllText($cmdPath, $cmdContent, (New-Object Text.UTF8Encoding $true))
Write-Output "桌面脚本: $cmdPath"

# ---------- 2) 开机自启动快捷方式 (隐藏窗口) ----------
$ws = New-Object -ComObject WScript.Shell
$lnkPath = Join-Path $startup 'Mio桌宠自启动.lnk'
$lnk = $ws.CreateShortcut($lnkPath)
$lnk.TargetPath = $pwshExe
$lnk.Arguments = '-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File "' + $startScript + '"'
$lnk.WorkingDirectory = 'C:\Users\xiao1\Mio'
if (Test-Path $icon) { $lnk.IconLocation = "$icon,0" }
$lnk.Save()
Write-Output "自启动快捷方式: $lnkPath"

# ---------- 3) 验证 ----------
Write-Output '--- 验证 ---'
Write-Output ("桌面脚本存在: " + (Test-Path $cmdPath))
Write-Output ("自启动存在: " + (Test-Path $lnkPath))
$lnk2 = $ws.CreateShortcut($lnkPath)
Write-Output ("自启动 Target: " + $lnk2.TargetPath)
Write-Output ("自启动 Args: " + $lnk2.Arguments)
